cmake_minimum_required(VERSION 3.16)
set(SUPPORTED_TARGETS esp32)
if(NOT EXISTS "$ENV{IDF_PATH}/tools/cmake/version.cmake")
    message(FATAL_ERROR "Export ESP-IDF v4.4.7 first")
endif()
include($ENV{IDF_PATH}/tools/cmake/version.cmake)
if(NOT "${IDF_VERSION_MAJOR}.${IDF_VERSION_MINOR}.${IDF_VERSION_PATCH}" STREQUAL "4.4.7")
    message(FATAL_ERROR "v0.1.1 requires exactly ESP-IDF v4.4.7 (EOL)")
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
set(PROJECT_VER "0.1.1")
project(EsperExperimental)
